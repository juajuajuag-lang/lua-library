local CameraBlockAction = "ProtoMenuCameraBlock"

local function BlockCamera(_, State)
    if State == Enum.UserInputState.Begin
    or State == Enum.UserInputState.Change then
        return Enum.ContextActionResult.Sink
    end

    return Enum.ContextActionResult.Pass

--drag

local Dragging = false
local DragStart
local StartPosition

local function BeginDrag(Input)
    Dragging = true
    DragStart = Input.Position
    StartPosition = MainFrame.Position

    ContextActionService:BindActionAtPriority(
        CameraBlockAction,
        BlockCamera,
        false,
        Enum.ContextActionPriority.High.Value,
        Enum.UserInputType.MouseMovement,
        Enum.UserInputType.MouseButton2,
        Enum.UserInputType.Touch
    )
end

local function UpdateDrag(Input)
    if not Dragging then
        return
    end

    local Delta = Input.Position - DragStart

    MainFrame.Position = UDim2.new(
        StartPosition.X.Scale,
        StartPosition.X.Offset + Delta.X,
        StartPosition.Y.Scale,
        StartPosition.Y.Offset + Delta.Y
    )
end

local function EndDrag()
    Dragging = false
    ContextActionService:UnbindAction(CameraBlockAction)
end

TopBar.InputBegan:Connect(function(Input)
    if Input.UserInputType == Enum.UserInputType.MouseButton1
    or Input.UserInputType == Enum.UserInputType.Touch then

        BeginDrag(Input)
    end
end)

UserInputService.InputChanged:Connect(function(Input)
    if Input.UserInputType == Enum.UserInputType.MouseMovement
    or Input.UserInputType == Enum.UserInputType.Touch then

        UpdateDrag(Input)
    end
end)

UserInputService.InputEnded:Connect(function(Input)
    if Input.UserInputType == Enum.UserInputType.MouseButton1
    or Input.UserInputType == Enum.UserInputType.Touch then

        EndDrag()
    end
end)

local function CreateSlider(
    Page,
    Text,
    Minimum,
    Maximum,
    Default,
    Step,
    Callback
)
    local Row = CreateRow(Page, 62)
    CreateLabel(Row, Text)

    local ValueLabel = Instance.new("TextLabel")
    ValueLabel.Name = "Value"
    ValueLabel.Size = UDim2.fromOffset(55, 18)

    ValueLabel.TextXAlignment = Enum.TextXAlignment.Right
    ValueLabel.Parent = Row

    local Track = Instance.new("Frame")
    Track.Name = "Track"
    Track.Size = UDim2.new(1, -20, 0, 5)
    Track.Position = UDim2.new(0, 10, 1, -16)
    Track.BackgroundColor3 = Color3.fromRGB(48, 48, 56)
    Track.BorderSizePixel = 0
    Track.Parent = Row

    local TrackCorner = Instance.new("UICorner")
    TrackCorner.CornerRadius = UDim.new(1, 0)
    TrackCorner.Parent = Track

    local Fill = Instance.new("Frame")
    Fill.Name = "Fill"
    Fill.Size = UDim2.new(0, 0, 1, 0)
    Fill.BackgroundColor3 = Color3.fromRGB(95, 145, 220)
    Fill.BorderSizePixel = 0
    Fill.Parent = Track

    local FillCorner = Instance.new("UICorner")
    FillCorner.CornerRadius = UDim.new(1, 0)
    FillCorner.Parent = Fill

    local Knob = Instance.new("Frame")
    Knob.Name = "Knob"
    Knob.Size = UDim2.fromOffset(13, 13)
    Knob.AnchorPoint = Vector2.new(0.5, 0.5)
    Knob.Position = UDim2.new(0, 0, 0.5, 0)
    Knob.BackgroundColor3 = Color3.fromRGB(235, 235, 240)
    Knob.BorderSizePixel = 0
    Knob.Parent = Track

    local KnobCorner = Instance.new("UICorner")
    KnobCorner.CornerRadius = UDim.new(1, 0)
    KnobCorner.Parent = Knob

    local Value = math.clamp(
        Default or Minimum,
        Minimum,
        Maximum
    )

    local DraggingSlider = false

    local function RoundValue(Number)
        if not Step or Step <= 0 then
            return Number
        end

        local Steps = math.floor(
            ((Number - Minimum) / Step) + 0.5
        )

        return Minimum + Steps * Step
    end

    local function FormatValue(Number)
        if Step and Step < 1 then
            return string.format("%.2f", Number)
        end

        return tostring(math.floor(Number + 0.5))
    end

    local function Update(Number, FireCallback)
        Value = math.clamp(
            RoundValue(Number),
            Minimum,
            Maximum
        )

        local Alpha

        if Maximum == Minimum then
            Alpha = 0
        else
            Alpha = (Value - Minimum)
                / (Maximum - Minimum)
        end

        Fill.Size = UDim2.new(
            Alpha,
            0,
            1,
            0
        )

        Knob.Position = UDim2.new(
            Alpha,
            0,
            0.5,
            0
        )

        ValueLabel.Text = FormatValue(Value)

        if FireCallback and Callback then
            Callback(Value)
        end
    end

    local function SetFromInput(Input)
        local X = Input.Position.X
        local StartX = Track.AbsolutePosition.X
        local Width = Track.AbsoluteSize.X

        if Width <= 0 then
            return
        end

        local Alpha = math.clamp(
            (X - StartX) / Width,
            0,
            1
        )

        local Number = Minimum
            + ((Maximum - Minimum) * Alpha)

        Update(Number, true)
    end

    Track.InputBegan:Connect(function(Input)
        if Input.UserInputType
            == Enum.UserInputType.MouseButton1
        or Input.UserInputType
            == Enum.UserInputType.Touch then

            DraggingSlider = true
            SetFromInput(Input)
        end
    end)

    UserInputService.InputChanged:Connect(function(Input)
        if not DraggingSlider then
            return
        end

        if Input.UserInputType
            == Enum.UserInputType.MouseMovement
        or Input.UserInputType
            == Enum.UserInputType.Touch then

            SetFromInput(Input)
        end
    end)

    UserInputService.InputEnded:Connect(function(Input)
        if Input.UserInputType
            == Enum.UserInputType.MouseButton1
        or Input.UserInputType
            == Enum.UserInputType.Touch then

            DraggingSlider = false
        end
    end)

    Update(Value, true)

    return {
        Row = Row,
        Track = Track,

        Get = function()
            return Value
        end,

        Set = function(Number)
            Update(Number, true)
        end
    }
end

local function IsInsideGui(Object, Position)
    if not Object or not Object.Visible then
        return false
    end

    local ObjectPosition = Object.AbsolutePosition
    local ObjectSize = Object.AbsoluteSize

    return Position.X >= ObjectPosition.X
        and Position.X <= ObjectPosition.X + ObjectSize.X
        and Position.Y >= ObjectPosition.Y
        and Position.Y <= ObjectPosition.Y + ObjectSize.Y
end

        )

        List.CanvasSize = UDim2.fromOffset(
            0,
            (#Options * 29) + 8
        )
    end

        if Callback then
            Callback(Value)
        end

        CloseDropdown()
    end

        local OptionButton = Instance.new("TextButton")
        OptionButton.Name = "Option"
        OptionButton.Size = UDim2.new(1, -8, 0, 26)

local function CreateMultiDropdown(
    Page,
    Text,
    Options,
    Defaults,
    Callback
)
    local Row = CreateRow(Page, 48)
    CreateLabel(Row, Text)

    local Button = Instance.new("TextButton")

    Container.Size = UDim2.fromOffset(145, 40)
    Container.BackgroundColor3 = Color3.fromRGB(31, 31, 38)
    Container.BorderSizePixel = 0
    Container.Visible = false
    Container.ZIndex = 100
    Container.Parent = ScreenGui

    local ContainerCorner = Instance.new("UICorner")
    ContainerCorner.CornerRadius = UDim.new(0, 6)
    ContainerCorner.Parent = Container

    local ContainerStroke = Instance.new("UIStroke")
    ContainerStroke.Color = Color3.fromRGB(58, 58, 68)
    ContainerStroke.Thickness = 1
    ContainerStroke.Parent = Container

    local List = Instance.new("ScrollingFrame")
    List.Name = "List"

    List.BorderSizePixel = 0
    List.ScrollBarThickness = 2
    List.ScrollBarImageColor3 = Color3.fromRGB(100, 100, 110)
    List.ZIndex = 101
    List.Parent = Container

    local Padding = Instance.new("UIPadding")
    Padding.PaddingTop = UDim.new(0, 4)
    Padding.PaddingBottom = UDim.new(0, 4)
    Padding.PaddingLeft = UDim.new(0, 4)
    Padding.PaddingRight = UDim.new(0, 4)
    Padding.Parent = List

    local Layout = Instance.new("UIListLayout")
    Layout.Padding = UDim.new(0, 3)
    Layout.SortOrder = Enum.SortOrder.LayoutOrder
    Layout.Parent = List

    local Selected = {}

    for _, Option in ipairs(Defaults or {}) do
        Selected[Option] = true
    end

    local OptionButtons = {}

    local function UpdateText()
        local Names = {}

        for _, Option in ipairs(Options) do
            if Selected[Option] then
                Names[#Names + 1] = tostring(Option)
            end
        end

        if #Names == 0 then
            ValueLabel.Text = "None"
        else
            ValueLabel.Text = table.concat(
                Names,
                ", "
            )
        end
    end

    local function FireCallback()
        if not Callback then
            return
        end

        local Result = {}

        for _, Option in ipairs(Options) do
            Result[Option] = Selected[Option] == true
        end

        Callback(Result)
    end

    local function UpdatePosition()
        if not Button.Visible then
            return
        end

        local Position = Button.AbsolutePosition
        local Size = Button.AbsoluteSize

        local X = Position.X
        local Y = Position.Y + Size.Y + 4

        local Camera = workspace.CurrentCamera

        if Camera then
            local Viewport = Camera.ViewportSize

            if X + 145 > Viewport.X then
                X = Viewport.X - 150
            end

            if Y + Container.AbsoluteSize.Y > Viewport.Y then
                Y = Position.Y - Container.AbsoluteSize.Y - 4
            end
        end

        Container.Position = UDim2.fromOffset(X, Y)
    end

    local function UpdateHeight()
        local Height = math.clamp(
            (#Options * 29) + 8,
            37,
            180
        )

        Container.Size = UDim2.fromOffset(
            145,
            Height
        )

        List.CanvasSize = UDim2.fromOffset(
            0,
            (#Options * 29) + 8
        )
    end

    local function UpdateOption(
        Option,
        OptionButton
    )
        if Selected[Option] then
            OptionButton.Text =
                "✓  " .. tostring(Option)

            OptionButton.BackgroundColor3 =
                Color3.fromRGB(55, 70, 62)
        else
            OptionButton.Text =
                tostring(Option)

            OptionButton.BackgroundColor3 =
                Color3.fromRGB(38, 38, 45)
        end
    end

    for Index, Option in ipairs(Options) do
        local OptionButton = Instance.new("TextButton")
        OptionButton.Name = "Option"
        OptionButton.Size = UDim2.new(1, -8, 0, 26)
        OptionButton.BackgroundColor3 = Color3.fromRGB(38, 38, 45)
        OptionButton.BorderSizePixel = 0
        OptionButton.TextColor3 = Color3.fromRGB(205, 205, 212)
        OptionButton.TextSize = 10
        OptionButton.Font = Enum.Font.Gotham
        OptionButton.AutoButtonColor = false
        OptionButton.ZIndex = 102
        OptionButton.LayoutOrder = Index
        OptionButton.Parent = List

        local OptionCorner = Instance.new("UICorner")
        OptionCorner.CornerRadius = UDim.new(0, 4)
        OptionCorner.Parent = OptionButton

        OptionButtons[Option] = OptionButton

        UpdateOption(
            Option,
            OptionButton
        )

        OptionButton.MouseEnter:Connect(function()
            if not Selected[Option] then
                OptionButton.BackgroundColor3 =
                    Color3.fromRGB(48, 48, 58)
            end
        end)

        OptionButton.MouseLeave:Connect(function()
            UpdateOption(
                Option,
                OptionButton
            )
        end)

        OptionButton.MouseButton1Click:Connect(function()
            Selected[Option] =
                not Selected[Option]

            UpdateOption(
                Option,
                OptionButton
            )

            UpdateText()
            FireCallback()
        end)
    end

    UpdateHeight()
    UpdateText()

    Button.MouseButton1Click:Connect(function()
        if OpenDropdown
        and OpenDropdown.Container == Container then

            CloseDropdown()
            return
        end

        CloseDropdown()

        UpdateHeight()
        UpdatePosition()

        Container.Visible = true

        OpenDropdown = {
            Container = Container,
            Button = Button,
            UpdatePosition = UpdatePosition
        }
    end)

    DropdownRegistry[#DropdownRegistry + 1] = {
        Container = Container,
        Button = Button,
        UpdatePosition = UpdatePosition
    }

    return {
        Row = Row,
        Button = Button,

        Get = function()
            local Result = {}

            for _, Option in ipairs(Options) do
                Result[Option] =
                    Selected[Option] == true
            end

            return Result
        end,

        Set = function(Values)
            if typeof(Values) ~= "table" then
                return
            end

            for _, Option in ipairs(Options) do
                Selected[Option] =
                    Values[Option] == true

                UpdateOption(
                    Option,
                    OptionButtons[Option]
                )
            end

            UpdateText()
            FireCallback()
        end
    }
end

        if Page.Visible then

            break
        end
    end

    if not ActivePage then
        ActivePage = "VISUAL"
        SelectPage(ActivePage)
    end

    for Name, Button in pairs(TabButtons) do
        if Name == ActivePage then
            Button.BackgroundColor3 =
                Color3.fromRGB(55, 75, 105)

            Button.TextColor3 =
                Color3.fromRGB(245, 245, 250)
        else
            Button.BackgroundColor3 =
                Color3.fromRGB(30, 30, 37)

            Button.TextColor3 =
                Color3.fromRGB(175, 175, 185)
        end
    end
end

local Players = game:GetService("Players")

local LocalPlayer = Players.LocalPlayer
local PlayerESPFolder = nil
local PlayerESPConnections = {}
local PlayerESPRunning = false

local function GetPlayerESPFolder()
    if PlayerESPFolder and PlayerESPFolder.Parent then
        return PlayerESPFolder
    end

    PlayerESPFolder = Instance.new("Folder")
    PlayerESPFolder.Name = "LocalPlayerESP"
    PlayerESPFolder.Parent = workspace

    return PlayerESPFolder
end

local function GetPlayerRole(Player)
    if not Player then
        return "Survivor"
    end

    local Team = Player.Team
    local TeamName = Team and Team.Name

    if TeamName then
        TeamName = string.lower(TeamName)

        if string.find(TeamName, "killer") then
            return "Killer"
        end

        if string.find(TeamName, "spect") then
            return "Spectator"
        end

        if string.find(TeamName, "survivor") then
            return "Survivor"
        end
    end

    return "Survivor"
end

local function IsRoleEnabled(Role)
    local Visual = Settings.Visual
    local Types = Visual.PlayerESPTypes

    if not Types then
        return true
    end

    if Role == "Killer" then
        return Types.Killer == true
    end

    if Role == "Spectator" then
        return Types.Spectator == true
    end

    return Types.Survivor == true
end

local function GetRoleColor(Role)
    local Visual = Settings.Visual

    if Role == "Killer" then
        return Visual.KillerColor or Color3.fromRGB(255, 0, 0)
    end

    if Role == "Spectator" then
        return Visual.SpectatorColor or Color3.fromRGB(255, 255, 255)
    end

    return Visual.SurvivorColor or Color3.fromRGB(0, 255, 0)
end

local function IsValidCharacter(Character)
    if not Character then
        return false
    end

    if not Character.Parent then
        return false
    end

    local Humanoid = Character:FindFirstChildOfClass("Humanoid")

    return Humanoid ~= nil
end

local function GetPlayerKey(Player)
    if not Player then
        return "Unknown"
    end

    return tostring(Player.UserId)
end

local function GetHighlightName(Player)
    return "Player_" .. GetPlayerKey(Player)
end

local function RemovePlayerESP(Player)
    if not PlayerESPFolder or not Player then
        return
    end

    local Highlight = PlayerESPFolder:FindFirstChild(
        GetHighlightName(Player)
    )

    if Highlight then
        Highlight:Destroy()
    end
end

local function RemoveAllPlayerESP()
    if not PlayerESPFolder then
        return
    end

    for _, Object in ipairs(PlayerESPFolder:GetChildren()) do
        Object:Destroy()
    end
end

local function CreatePlayerHighlight(Player)
    if not Player or Player == LocalPlayer then
        return
    end

    if not Settings.Visual.EnablePlayerESP then
        RemovePlayerESP(Player)
        return
    end

    local Character = Player.Character

    if not IsValidCharacter(Character) then
        RemovePlayerESP(Player)
        return
    end

    local Role = GetPlayerRole(Player)

    if not IsRoleEnabled(Role) then
        RemovePlayerESP(Player)
        return
    end

    local Folder = GetPlayerESPFolder()
    local Name = GetHighlightName(Player)
    local Highlight = Folder:FindFirstChild(Name)

    if not Highlight then
        Highlight = Instance.new("Highlight")
        Highlight.Name = Name
        Highlight.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
        Highlight.Parent = Folder
    end

    local Color = GetRoleColor(Role)

    Highlight.Adornee = Character
    Highlight.FillColor = Color
    Highlight.OutlineColor = Color
    Highlight.FillTransparency =
        Settings.Visual.ESPFillTransparency or 0.95
    Highlight.OutlineTransparency =
        Settings.Visual.ESPOutlineTransparency or 0.3
    Highlight.Enabled = true
end

local function RefreshPlayerESP()
    if not Settings.Visual.EnablePlayerESP then
        RemoveAllPlayerESP()
        return
    end

    for _, Player in ipairs(Players:GetPlayers()) do
        if Player ~= LocalPlayer then
            pcall(CreatePlayerHighlight, Player)
        end
    end
end

local function WatchPlayer(Player)
    if Player == LocalPlayer then
        return
    end

    if PlayerESPConnections[Player] then
        for _, Connection in ipairs(PlayerESPConnections[Player]) do
            pcall(function()
                Connection:Disconnect()
            end)
        end
    end

    PlayerESPConnections[Player] = {}

    table.insert(
        PlayerESPConnections[Player],
        Player.CharacterAdded:Connect(function()
            RemovePlayerESP(Player)

            task.delay(0.15, function()
                if Settings.Visual.EnablePlayerESP then
                    pcall(CreatePlayerHighlight, Player)
                end
            end)
        end)
    )

    table.insert(
        PlayerESPConnections[Player],
        Player.CharacterRemoving:Connect(function()
            RemovePlayerESP(Player)
        end)
    )

    table.insert(
        PlayerESPConnections[Player],
        Player:GetPropertyChangedSignal("Team"):Connect(function()
            RemovePlayerESP(Player)

            if Settings.Visual.EnablePlayerESP then
                pcall(CreatePlayerHighlight, Player)
            end
        end)
    )
end

local function UnwatchPlayer(Player)
    RemovePlayerESP(Player)

    local Connections = PlayerESPConnections[Player]

    if Connections then
        for _, Connection in ipairs(Connections) do
            pcall(function()
                Connection:Disconnect()
            end)
        end
    end

    PlayerESPConnections[Player] = nil
end

for _, Player in ipairs(Players:GetPlayers()) do
    WatchPlayer(Player)
end

Players.PlayerAdded:Connect(function(Player)
    WatchPlayer(Player)

    if Settings.Visual.EnablePlayerESP then
        task.delay(0.15, function()
            pcall(CreatePlayerHighlight, Player)
        end)
    end
end)

Players.PlayerRemoving:Connect(function(Player)
    UnwatchPlayer(Player)
end)

task.spawn(function()
    local LastEnabled = nil
    local LastSurvivor = nil
    local LastKiller = nil
    local LastSpectator = nil
    local LastFill = nil
    local LastOutline = nil

    while true do
        local Visual = Settings.Visual

        local Enabled = Visual.EnablePlayerESP == true
        local Types = Visual.PlayerESPTypes or {}

        local Survivor = Types.Survivor == true
        local Killer = Types.Killer == true
        local Spectator = Types.Spectator == true

        local Fill = Visual.ESPFillTransparency
        local Outline = Visual.ESPOutlineTransparency

        local Changed =
            Enabled ~= LastEnabled
            or Survivor ~= LastSurvivor
            or Killer ~= LastKiller
            or Spectator ~= LastSpectator
            or Fill ~= LastFill
            or Outline ~= LastOutline

        if Changed then
            LastEnabled = Enabled
            LastSurvivor = Survivor
            LastKiller = Killer
            LastSpectator = Spectator
            LastFill = Fill
            LastOutline = Outline

            if Enabled then
                RefreshPlayerESP()
            else
                RemoveAllPlayerESP()
            end
        elseif Enabled then
            RefreshPlayerESP()
        end

        task.wait(0.25)
    end
end)

task.spawn(function()
    local LastSurvivor = nil
    local LastKiller = nil
    local LastSpectator = nil

    while true do
        local Visual = Settings.Visual
        local Types = Visual.PlayerESPTypes or {}

        local Survivor = Types.Survivor == true
        local Killer = Types.Killer == true
        local Spectator = Types.Spectator == true

        if Survivor ~= LastSurvivor
        or Killer ~= LastKiller
        or Spectator ~= LastSpectator then
            LastSurvivor = Survivor
            LastKiller = Killer
            LastSpectator = Spectator

            if Visual.EnablePlayerESP then
                RefreshPlayerESP()
            end
        end

        task.wait(0.1)
    end
end)

local ReplicatedStorage = game:GetService("ReplicatedStorage")

local SurvivorItemNames = {
    ["Motion Tracker"] = true,
    ["Gate"] = true,
    ["Flashlight"] = true,
    ["Bandage"] = true,
    ["Parrying Dagger"] = true,
    ["Adrenaline Shot"] = true,
    ["Twist of Fate"] = true,
    ["Shadow Clone"] = true,
    ["Holy Water"] = true,
    ["WaxBound Candle"] = true,
    ["Riot Shield"] = true,
    ["Emperor"] = true,
    ["AWP"] = true
}

local function GetSurvivorItem(Player)
    local Character = Player.Character

    if not Character then
        return nil
    end

    for _, Object in ipairs(Character:GetDescendants()) do
        if Object:IsA("Tool")
        or Object:IsA("Accessory")
        or Object:IsA("Model") then
            if SurvivorItemNames[Object.Name] then
                return Object.Name
            end
        end
    end

    return nil
end

local function GetItemImageId(ItemName)
    local ItemsFolder = ReplicatedStorage:FindFirstChild("Items")

    if not ItemsFolder then
        return nil
    end

    local ItemObject = ItemsFolder:FindFirstChild(ItemName)

    if not ItemObject then
        return nil
    end

    if ItemObject:IsA("Decal")
    or ItemObject:IsA("Texture") then
        return ItemObject.Texture
    end

    local Texture =
        ItemObject:FindFirstChildWhichIsA("Decal", true)
        or ItemObject:FindFirstChildWhichIsA("Texture", true)

    if Texture then
        return Texture.Texture
    end

    local NamedTexture =
        ItemObject:FindFirstChild("Texture", true)

    if NamedTexture
    and (NamedTexture:IsA("Decal")
    or NamedTexture:IsA("Texture")) then
        return NamedTexture.Texture
    end

    return nil
end

local function RemoveSurvivorItemESP(Player)
    if not Player then
        return
    end

    local Folder = PlayerESPFolder

    if not Folder then
        return
    end

    local Name = "PlayerItem_" .. GetPlayerKey(Player)
    local Icon = Folder:FindFirstChild(Name)

    if Icon then
        Icon:Destroy()
    end
end

local function UpdateSurvivorItemESP(Player)
    if not Player or Player == LocalPlayer then
        return
    end

    local Types = Settings.Visual.PlayerESPTypes or {}

    if not Types.SurvivorItems then
        RemoveSurvivorItemESP(Player)
        return
    end

    if GetPlayerRole(Player) ~= "Survivor" then
        RemoveSurvivorItemESP(Player)
        return
    end

    local Character = Player.Character

    if not IsValidCharacter(Character) then
        RemoveSurvivorItemESP(Player)
        return
    end

    local Torso =
        Character:FindFirstChild("HumanoidRootPart")
        or Character:FindFirstChild("UpperTorso")
        or Character:FindFirstChild("Torso")

    if not Torso or not Torso:IsA("BasePart") then
        RemoveSurvivorItemESP(Player)
        return
    end

    local ItemName = GetSurvivorItem(Player)

    if not ItemName then
        RemoveSurvivorItemESP(Player)
        return
    end

    local ImageId = GetItemImageId(ItemName)

    if not ImageId then
        RemoveSurvivorItemESP(Player)
        return
    end

    local Folder = GetPlayerESPFolder()
    local Name = "PlayerItem_" .. GetPlayerKey(Player)
    local Icon = Folder:FindFirstChild(Name)

    if not Icon then
        Icon = Instance.new("BillboardGui")
        Icon.Name = Name
        Icon.AlwaysOnTop = true
        Icon.LightInfluence = 0
        Icon.MaxDistance = 0
        Icon.Size = UDim2.fromOffset(20, 20)
        Icon.StudsOffset = Vector3.new(0, 0, -1.6)
        Icon.Parent = Folder

        local Image = Instance.new("ImageLabel")
        Image.Name = "ImageLabel"
        Image.BackgroundTransparency = 1
        Image.Size = UDim2.fromScale(1, 1)
        Image.Parent = Icon
    end

    Icon.Adornee = Torso
    Icon.Enabled = true

    local Image = Icon:FindFirstChild("ImageLabel")

    if Image then
        Image.Image = ImageId
    end
end

task.spawn(function()
    local LastState = nil

    while true do
        local Types = Settings.Visual.PlayerESPTypes or {}
        local Enabled = Types.SurvivorItems == true

        if Enabled ~= LastState then
            LastState = Enabled

            for _, Player in ipairs(Players:GetPlayers()) do
                if Player ~= LocalPlayer then
                    UpdateSurvivorItemESP(Player)
                end
            end
        elseif Enabled and Settings.Visual.EnablePlayerESP then
            for _, Player in ipairs(Players:GetPlayers()) do
                if Player ~= LocalPlayer then
                    UpdateSurvivorItemESP(Player)
                end
            end
        end

        task.wait(0.25)
    end
end)

        Label.Font = Enum.Font.Gotham
        Label.TextStrokeTransparency = 0.65
        Label.TextStrokeColor3 = Color3.new(0, 0, 0)
        Label.Parent = Tag
    end

    Label.TextSize = Settings.Visual.ESPTextSize or 12
    Label.TextColor3 = GetRoleColor(Role)
    Label.Text = Player.Name

    Tag.Adornee = Head
    Tag.Enabled = true
end

local function RefreshAllPlayerNametags()
    if not Settings.Visual.EnablePlayerESP then
        for _, Player in ipairs(Players:GetPlayers()) do
            RemovePlayerNametag(Player)
        end
        return
    end

    for _, Player in ipairs(Players:GetPlayers()) do
        if Player ~= LocalPlayer then
            UpdatePlayerNametag(Player)
        end
    end
end

task.spawn(function()
    local LastEnabled = nil

    while true do
        local Types = Settings.Visual.PlayerESPTypes or {}
        local Enabled =
            Settings.Visual.EnablePlayerESP == true
            and Types.Nametags == true

        if Enabled ~= LastEnabled then
            LastEnabled = Enabled
            RefreshAllPlayerNametags()
        elseif Enabled then
            RefreshAllPlayerNametags()
        end

        task.wait(0.25)
    end
end)

local function UpdatePlayerDistanceESP(Player)
    if not Player or Player == LocalPlayer then
        return
    end

    local Types = Settings.Visual.PlayerESPTypes or {}

    if not Types.DistanceESP then
        return
    end

    if not Settings.Visual.EnablePlayerESP then
        return
    end

    local Character = Player.Character

    if not IsValidCharacter(Character) then
        return
    end

    local MyCharacter = LocalPlayer.Character

    if not MyCharacter then
        return
    end

    local MyRoot = MyCharacter:FindFirstChild("HumanoidRootPart")
    local TargetRoot = Character:FindFirstChild("HumanoidRootPart")

    if not MyRoot or not TargetRoot then
        return
    end

    local Distance = math.floor(
        (MyRoot.Position - TargetRoot.Position).Magnitude
    )

    local NameTag = PlayerESPFolder
        and PlayerESPFolder:FindFirstChild(
            "PlayerTag_" .. GetPlayerKey(Player)
        )

    if not NameTag then
        UpdatePlayerNametag(Player)
        NameTag = PlayerESPFolder
            and PlayerESPFolder:FindFirstChild(
                "PlayerTag_" .. GetPlayerKey(Player)
            )
    end

    if not NameTag then
        return
    end

    local Label = NameTag:FindFirstChild("NameLabel")

    if not Label then
        return
    end

    local ShowName = Types.Nametags == true

    if ShowName then
        Label.Text = Player.Name .. " [" .. Distance .. "m]"
    else
        Label.Text = "[" .. Distance .. "m]"
    end

    Label.TextSize = Settings.Visual.ESPTextSize or 12
end

task.spawn(function()
    while true do
        local Types = Settings.Visual.PlayerESPTypes or {}

        if Settings.Visual.EnablePlayerESP
        and Types.DistanceESP then
            for _, Player in ipairs(Players:GetPlayers()) do
                if Player ~= LocalPlayer then
                    UpdatePlayerDistanceESP(Player)
                end
            end
        end

        task.wait(0.1)
    end
end)

local KillerWarningYellow = "rbxassetid://120753562298647"
local KillerWarningRed = "rbxassetid://99499680507460"

local KillerWarningObjects = {}

local function ClearKillerWarning(Player)
    local Warning = KillerWarningObjects[Player]

    if Warning then
        Warning:Destroy()
        KillerWarningObjects[Player] = nil
    end
end

local function CreateKillerWarning(Player)
    if not Player or Player == LocalPlayer then
        return
    end

    local Character = Player.Character
    if not Character then
        return
    end

    local Root = Character:FindFirstChild("HumanoidRootPart")
    local Humanoid = Character:FindFirstChildOfClass("Humanoid")

    if not Root or not Humanoid or Humanoid.Health <= 0 then
        return
    end

    local Warning = KillerWarningObjects[Player]

    if Warning and Warning.Parent ~= Root then
        Warning:Destroy()
        Warning = nil
        KillerWarningObjects[Player] = nil
    end

    if not Warning then
        Warning = Instance.new("BillboardGui")
        Warning.Name = "LocalKillerWarning"
        Warning.Adornee = Root
        Warning.AlwaysOnTop = true
        Warning.LightInfluence = 0
        Warning.MaxDistance = 2000
        Warning.Parent = Root

        local Yellow = Instance.new("ImageLabel")
        Yellow.Name = "YellowWarning"
        Yellow.BackgroundTransparency = 1
        Yellow.Image = KillerWarningYellow
        Yellow.Size = UDim2.fromScale(1, 1)
        Yellow.Position = UDim2.fromScale(0, 0)
        Yellow.Parent = Warning

        local Red = Instance.new("ImageLabel")
        Red.Name = "RedWarning"
        Red.BackgroundTransparency = 1
        Red.Image = KillerWarningRed
        Red.Size = UDim2.fromScale(0.62, 1)
        Red.Position = UDim2.fromScale(0.38, 0)
        Red.Visible = false
        Red.Parent = Warning

        KillerWarningObjects[Player] = Warning
    end

    return Warning
end

local function UpdateKillerWarning(Player, Distance)
    local Warning = CreateKillerWarning(Player)

    if not Warning then
        return
    end

    local Yellow = Warning:FindFirstChild("YellowWarning")
    local Red = Warning:FindFirstChild("RedWarning")

    local Danger = Distance <= 40

    if Danger then
        Warning.Size = UDim2.fromOffset(76, 56)

        if Yellow then
            Yellow.Size = UDim2.fromScale(0.62, 1)
            Yellow.Position = UDim2.fromScale(0, 0)
            Yellow.Visible = true
        end

        if Red then
            Red.Size = UDim2.fromScale(0.62, 1)
            Red.Position = UDim2.fromScale(0.38, 0)
            Red.Visible = true
        end
    else
        Warning.Size = UDim2.fromOffset(56, 56)

        if Yellow then
            Yellow.Size = UDim2.fromScale(1, 1)
            Yellow.Position = UDim2.fromScale(0, 0)
            Yellow.Visible = true
        end

        if Red then
            Red.Visible = false
        end
    end
end

local function GetNearestKillerDistance(Player)
    local Character = Player.Character

    if not Character then
        return math.huge
    end

    local Root = Character:FindFirstChild("HumanoidRootPart")
    local Humanoid = Character:FindFirstChildOfClass("Humanoid")

    if not Root or not Humanoid or Humanoid.Health <= 0 then
        return math.huge
    end

    local Nearest = math.huge

    for _, Killer in ipairs(Players:GetPlayers()) do
        if Killer ~= LocalPlayer then
            local KillerCharacter = Killer.Character

            if KillerCharacter and GetPlayerRole(Killer) == "Killer" then
                local KillerRoot =
                    KillerCharacter:FindFirstChild("HumanoidRootPart")

                local KillerHumanoid =
                    KillerCharacter:FindFirstChildOfClass("Humanoid")

                if KillerRoot
                and KillerHumanoid
                and KillerHumanoid.Health > 0 then
                    local Distance =
                        (Root.Position - KillerRoot.Position).Magnitude

                    if Distance < Nearest then
                        Nearest = Distance
                    end
                end
            end
        end
    end

    return Nearest
end

task.spawn(function()
    while true do
        local Enabled = Settings.Visual.SurvivorKillerWarning == true

        if Enabled then
            for _, Player in ipairs(Players:GetPlayers()) do
                if Player ~= LocalPlayer
                and GetPlayerRole(Player) == "Survivor" then
                    local Distance = GetNearestKillerDistance(Player)

                    if Distance <= 60 then
                        UpdateKillerWarning(Player, Distance)
                    else
                        ClearKillerWarning(Player)
                    end
                else
                    ClearKillerWarning(Player)
                end
            end
        else
            for Player in pairs(KillerWarningObjects) do
                ClearKillerWarning(Player)
            end
        end

        task.wait(0.15)
    end
end)

Players.PlayerRemoving:Connect(function(Player)
    ClearKillerWarning(Player)
end)
